[ROOT]
|> app
    |> Models [devCMD-6] (hier komen de models die met de database interacten)
    |> http
        |> Controllers [devCMD-7] (hier komen controlers die php runnen)
        
|> database
    |> migrations [devCMD-8] (hier komen migrations die de database maken)

|> docs (een folder met alle documenten die we gebruiken om de site te maken)
    |> info (deze docs)

|> public (root on server end)
    |> build (word gemaakt zodat alle nodige files kunnen gebruikt worden op de live)
    |> images [devCMD-5] (plek waar alle fotos inkomen)
    |> index.php (index on sever end)
    |> hot (blocks css on live, REMOVE IT)
    |> favicon.ico (de image die op te tab komt)
    
|> resources (css, js en pages)
    |> css [devCMD-1]
        |> app.css (globol css file)
        |> [PAGE].css (css die alleen op [PAGE] word gebruikt)

    |> js [devCMD-1]
        |> app.js (globol js file)
        |> [PAGE].js (js die alleen op [PAGE] word gebruikt)

    |> views [devCMD-3] (de PHP paginas)
        |> include [devCMD-4] (dingen die we kunnen include bv. popups)
        |> layouts [devCMD-4] (de layouts die we kunnen gebruiken)
        |> [PAGE].blade.php (je moet .blade toevoegen)
        
|> routes
    |> web.php [devCMD-2] (de pagina die alle paginas gaat re route)

|> .env (all prived info, DONT UPLOAD EVER)
|> .env.example (dit is een voorbeeld, deze word geupload dus type hier geen info)
|> .gitignore (files die niet naar git worden geupload. bv. env)
